  function el(tag, className, text) {
    const node = document.createElement(tag)
    if (className) node.className = className
    if (text !== undefined) node.textContent = text
    return node
  }

  function svg(tag, attributes) {
    const node = document.createElementNS(SVG, tag)
    Object.entries(attributes || {}).forEach((pair) => node.setAttribute(pair[0], pair[1]))
    return node
  }

  function append(parent, ...children) {
    children.filter(Boolean).forEach((child) => parent.append(child))
    return parent
  }

  // Agent text may mark code with backticks; everything is inserted as text, never as HTML.
  function inline(parent, text) {
    String(text || "")
      .split(/(`[^`]+`)/)
      .forEach((piece) => {
        if (piece.length > 2 && piece.startsWith("`") && piece.endsWith("`")) parent.append(el("code", "", piece.slice(1, -1)))
        else if (piece) parent.append(document.createTextNode(piece.replace(/\*\*/g, "")))
      })
    return parent
  }

  function plural(count, word) {
    return count + " " + word + (count === 1 ? "" : "s")
  }

  function duration(ms) {
    const seconds = Math.round((ms || 0) / 1000)
    if (seconds < 60) return seconds + " s"
    const minutes = Math.floor(seconds / 60)
    if (minutes < 60) return minutes + " min " + (seconds % 60) + " s"
    return Math.floor(minutes / 60) + " h " + (minutes % 60) + " min"
  }

  function key(entry) {
    return entry.number + "." + entry.revision
  }

  function label(entry) {
    return "Checkpoint " + entry.number + (entry.revision > 1 ? ", revision " + entry.revision : "")
  }

  function short(entry) {
    return entry.revision > 1 ? entry.number + "r" + entry.revision : String(entry.number)
  }

  function status(entry) {
    return entry.status === "waiting" ? "waiting" : entry.decision
  }

  function counts(additions, deletions) {
    return append(el("span", "counts"), el("span", "a", "+" + additions), document.createTextNode(" "), el("span", "d", "−" + deletions))
  }

  function tag(kind, text) {
    return el("span", "tag " + kind, text || (kind === "claim" ? "the agent's words · not verified" : "computed from the code"))
  }

  function part(title, ...tags) {
    const section = el("section", "part")
    section.append(append(el("header"), el("h2", "", title), ...tags))
    return section
  }

  // Agents write paths in different ways, such as "./src/a.ts" or just "a.ts".
  function same(a, b) {
    const clean = (value) => String(value).trim().replace(/\\/g, "/").replace(/^\.\//, "")
    const left = clean(a)
    const right = clean(b)
    return left === right || left.endsWith("/" + right) || right.endsWith("/" + left)
  }

  function sentence(items) {
    return items.length > 1 ? items.slice(0, -1).join(", ") + " and " + items.at(-1) : items.join("")
  }

  // A function as "total()", a method as "Cart.total()", and a class or type with its kind.
  function named(symbol) {
    if (symbol.kind === "method") return (symbol.parent ? symbol.parent + "." : "") + symbol.name + "()"
    if (symbol.kind === "function") return symbol.name + "()"
    if (symbol.kind === "class" || symbol.kind === "type") return symbol.kind + " " + symbol.name
    return symbol.name
  }

  // The changed or removed functions an affected file mentions, with how each one changed.
  function mentioned(entry, item) {
    return item.names.map((name) => {
      const found = entry.files
        .filter((file) => item.uses.includes(file.file))
        .flatMap((file) => file.symbols || [])
        .find((symbol) => symbol.name === name && symbol.change !== "added")
      return found ? named(found) + " (" + CHANGES[found.change] + ")" : name
    })
  }

  // Opens a file's full diff at the blocks where one of its functions changed, in the view `go` shows it in.
  function focus(id, file, index, go) {
    state.open.add(id)
    state.mark = { id, index }
    go(file)
    requestAnimationFrame(() => {
      const target = document.querySelector("tr.where.mark")
      if (target) target.scrollIntoView({ behavior: "smooth", block: "center" })
    })
  }

  function choose(next) {
    const list = state.data.checkpoints
    state.selected = next
    state.follow = next === key(list.at(-1))
    history.replaceState(null, "", "#" + next)
    render()
  }

  // Selects a file in the project tree, opening the folders on the way.
  function select(file) {
    state.file = file
    file
      .split("/")
      .slice(0, -1)
      .forEach((_, index, parts) => state.folders.set(parts.slice(0, index + 1).join("/") + "/", true))
  }

  // Shows a file in the project tree and its details. The scroll waits for the frame in which render() puts the page
  // back where it was, or that would stop it.
  function pick(file, scroll) {
    select(file)
    render()
    if (scroll) requestAnimationFrame(() => document.getElementById("project")?.scrollIntoView({ behavior: "smooth", block: "start" }))
  }

  // Opens the side panel next to the flowchart for a box, or for the file an arrow points at; the same click again
  // closes it. The project tree selects the same file, so both views stay on it.
  function show(id, step) {
    const again = state.panel === id && state.step === step
    state.panel = again ? undefined : id
    state.step = again ? undefined : step
    state.reveal = !again
    if (!again && known(id, state.data.checkpoints)) select(id)
    render()
  }

  function hide() {
    state.panel = undefined
    state.step = undefined
    render()
  }

  // Live updates wait while the user types in a box on the page, so it is not redrawn under the cursor. The chat is not
  // part of the page, so the page keeps updating while the user writes to the agent.
  function typing() {
    const active = document.activeElement
    return Boolean(active && active.closest("#page") && active.matches("input[type=text], textarea"))
  }

  function refresh() {
    if (typing()) state.stale = true
    else render()
  }

  function render() {
    state.stale = false
    const data = state.data
    const page = document.getElementById("page")
    const y = window.scrollY
    // The panel keeps its own scroll position while the page updates, unless it now shows another file.
    const before = document.querySelector(".panel")
    const kept = before && before.dataset.file === state.panel ? before.scrollTop : 0
    // So does a diagram that scrolls sideways.
    const left = document.querySelector(".machine")?.scrollLeft || 0
    document.title = "Understand map · " + data.session.title
    document.getElementById("session").textContent = data.session.title
    const list = data.checkpoints
    if (wanted && list.some((entry) => key(entry) === wanted)) {
      state.selected = wanted
      state.follow = wanted === key(list.at(-1))
      wanted = undefined
    }
    if (state.follow || !list.some((entry) => key(entry) === state.selected))
      state.selected = list.length ? key(list.at(-1)) : undefined
    talk()
    page.replaceChildren()
    // The files the agent changed since its last checkpoint, as it changes them; each opens its live diff.
    if (data.unreviewed.length) {
      const banner = el("p", "working", data.working ? "The agent is working. Changed so far, for its next checkpoint: " : "Changed since the last checkpoint and not reviewed yet: ")
      data.unreviewed.forEach((file, index) => {
        const link = el("button", "toggle", file.file)
        link.addEventListener("click", () => show(file.file))
        append(banner, index ? document.createTextNode(", ") : undefined, link)
      })
      page.append(banner)
    }
    const entry = list.find((item) => key(item) === state.selected)
    if (!entry) {
      page.append(el("p", "empty", "No checkpoint yet. The map fills in when the agent finishes its first feature."))
      return
    }
    // Each checkpoint opens on its own first changed file, with the flowchart's panel closed.
    if (state.at !== key(entry)) {
      state.panel = undefined
      state.step = undefined
    }
    if (state.at !== key(entry) || !state.file || !known(state.file, list)) state.file = entry.files[0] && entry.files[0].file
    state.at = key(entry)
    page.append(pills(list), hero(entry))
    if (entry.overview) {
      const big = part("The big picture", tag("claim"))
      big.append(inline(el("p", "prose"), entry.overview))
      page.append(big)
    }
    append(page, machine(entry, data.project), risks(entry), project(entry, data.project))
    if (entry.check) {
      const tryIt = part("How to try it", tag("claim"))
      tryIt.append(append(el("div", "try"), inline(el("p"), entry.check)))
      page.append(tryIt)
    }
    page.append(did(entry), session(data.stats, list))
    requestAnimationFrame(() => {
      layout()
      window.scrollTo(0, y)
      const panel = document.querySelector(".panel")
      if (panel) panel.scrollTop = kept
      const area = document.querySelector(".machine")
      if (area) area.scrollLeft = left
      if (state.reveal) reveal(area, panel)
      state.reveal = false
    })
  }

  // After a click opens the panel, the diagram is narrower: it scrolls sideways to keep the lit boxes in view, or the
  // box the panel shows when they do not fit. On a narrow screen the panel sits below the diagram and scrolls into view.
  function reveal(area, panel) {
    const lit = area ? [...area.querySelectorAll(".node.lit")] : []
    if (lit.length) {
      const first = Math.min(...lit.map((box) => box.offsetLeft))
      const last = Math.max(...lit.map((box) => box.offsetLeft + box.offsetWidth))
      const shown = lit.find((box) => box.dataset.id === state.panel) || lit[0]
      const middle = last - first <= area.clientWidth ? (first + last) / 2 : shown.offsetLeft + shown.offsetWidth / 2
      area.scrollLeft = middle - area.clientWidth / 2
    }
    if (!panel || !window.matchMedia("(max-width: 900px)").matches) return
    const top = panel.getBoundingClientRect().top + window.scrollY - document.querySelector(".top").offsetHeight - 8
    window.scrollTo({ top, behavior: "smooth" })
  }

  function known(file, list) {
    return list.some((entry) => entry.files.some((item) => item.file === file)) || state.data.project.includes(file)
  }

  function pills(list) {
    const row = el("nav", "pills")
    row.setAttribute("aria-label", "Checkpoints")
    list.forEach((entry) => {
      const pill = el("button", "pill" + (key(entry) === state.selected ? " selected" : "") + (entry.status === "waiting" ? " waiting" : ""))
      append(pill, el("b", "", short(entry)), el("span", "", entry.title), el("span", "state", DECISIONS[status(entry)]))
      pill.addEventListener("click", () => choose(key(entry)))
      row.append(pill)
    })
    return row
  }

  function hero(entry) {
    const eyebrow = append(el("div", "eyebrow", label(entry)), el("span", "status " + status(entry), DECISIONS[status(entry)]))
    return append(el("div", "hero"), eyebrow, el("h1", "", entry.title), inline(el("p", "lead"), entry.impact), entry.status === "waiting" ? decide(entry) : undefined)
  }

  // Approve or stop the checkpoint waiting for the user, as its options in the terminal do. Anything the user writes in
  // the chat instead becomes their revision request.
  function decide(entry) {
    const approve = el("button", "button primary", state.answering ? "Sending…" : "Approve")
    const stop = el("button", "button", "Stop")
    approve.disabled = state.answering
    stop.disabled = state.answering
    approve.addEventListener("click", () => answer("approve"))
    stop.addEventListener("click", () => {
      if (window.confirm("Stop the run at " + label(entry) + "?")) answer("stop")
    })
    const other = el("button", "toggle", "tell the agent what to change")
    other.addEventListener("click", () => toggleChat(true))
    return append(el("div", "decide"), el("strong", "", label(entry) + " is waiting for your answer."), approve, stop, append(el("span", "muted"), document.createTextNode("or "), other))
  }

  async function answer(decision) {
    state.answering = true
    refresh()
    const result = await post("answer", { decision })
    state.answering = false
    document.getElementById("chat-note").textContent = result.ok
      ? decision === "approve" ? "Approved: the agent continues." : "Stopped: the run ends here."
      : result.body.error || "OpenCode did not take the answer."
    refresh()
  }

  // The chat with the agent shows the same conversation as the terminal. It is not part of the page, so redrawing the
  // page never touches what the user is writing to the agent.
  function talk() {
    const data = state.data
    const waiting = data.checkpoints.find((entry) => entry.status === "waiting")
    document.getElementById("chat-dot").className = "dot" + (waiting ? " wait" : data.working ? " on" : "")
    document.getElementById("chat-state").textContent = waiting
      ? label(waiting) + " is waiting for your answer"
      : data.working ? "The agent is working…" : "The agent is waiting for your next request"
    // A list the user has read to the end keeps showing the newest line. It is scrolled last, because the bar that
    // approves a checkpoint changes how tall the list is.
    const list = document.getElementById("chat-list")
    const end = list.scrollTop + list.clientHeight >= list.scrollHeight - 40
    list.replaceChildren(...data.chat.map((item) => (item.role === "status" ? el("p", "msg event", item.text) : inline(el("p", "msg " + item.role), item.text))))
    document.getElementById("chat-decide").replaceChildren(...(waiting ? [decide(waiting)] : []))
    document.getElementById("chat-text").placeholder = waiting ? "Tell the agent what to change in " + label(waiting) : "Tell the agent what to change"
    if (end) list.scrollTop = list.scrollHeight
  }

  function toggleChat(open) {
    state.chat = open
    document.getElementById("chat").hidden = !open
    document.getElementById("chat-open").setAttribute("aria-expanded", String(open))
    if (!open) return
    const list = document.getElementById("chat-list")
    list.scrollTop = list.scrollHeight
    document.getElementById("chat-text").focus()
  }

  // Sends what the user wrote to the agent: the answer to a waiting checkpoint, or else a new message in the session.
  async function ask() {
    const box = document.getElementById("chat-text")
    const note = document.getElementById("chat-note")
    const send = document.getElementById("chat-send")
    const text = box.value.trim()
    if (!text || state.sending) return
    state.sending = true
    send.disabled = true
    note.textContent = "Sending…"
    const result = await post("send", { text })
    state.sending = false
    send.disabled = false
    if (result.ok) box.value = ""
    note.textContent = !result.ok
      ? result.body.error || "OpenCode did not take the message."
      : result.body.via === "revision"
        ? "Sent as your answer to the waiting checkpoint: the agent revises it, and the map follows its changes."
        : "Sent: the map follows the agent's changes as it works."
  }

  // The flowchart: each file the program passes through is a box, and each step is a numbered arrow.
  function machine(entry, files) {
    const section = part("How it works", tag("claim", "drawn by the model from the diff · not verified"), tag("fact", "arrows checked against the code"))
    const flow = entry.flow || []
    if (!flow.length) {
      const empty = el("p", "muted", "No flowchart for this change: the model did not draw one.")
      section.append(state.panel ? append(el("div", "flow open"), empty, panel(entry, [], undefined)) : empty)
      return section
    }
    const resolve = (name) => {
      if (/^(user|you)$/i.test(String(name).trim())) return { id: "user", name: "You", kind: "user" }
      const changed = entry.files.find((file) => same(file.file, name))
      if (changed)
        return { id: changed.file, name: changed.file.split("/").at(-1), kind: changed.status, file: changed.file, parts: (changed.symbols || []).map(named) }
      const existing = files.find((file) => same(file, name))
      const risky = (entry.affected || []).some((item) => item.file === existing)
      if (existing) return { id: existing, name: existing.split("/").at(-1), kind: risky ? "affected" : "context", file: existing }
      return { id: String(name), name: String(name).split("/").at(-1), kind: "missing" }
    }
    const nodes = []
    const steps = flow.map((step, index) => {
      const from = resolve(step.from)
      const to = resolve(step.to)
      ;[from, to].forEach((node) => {
        if (!nodes.some((item) => item.id === node.id)) nodes.push(node)
      })
      return { number: index + 1, from: from.id, to: to.id, action: step.action, confirmed: step.confirmed }
    })
    // Boxes that share a file name, such as routes/reports.js and services/reports.js, also show their folder. A box
    // for a file the agent is changing now is marked, so the diagram follows the agent's work until its next checkpoint.
    const counts = nodes.reduce((all, node) => all.set(node.name, (all.get(node.name) || 0) + 1), new Map())
    nodes.forEach((node) => {
      if (node.kind !== "user" && counts.get(node.name) > 1) node.folder = String(node.id).split("/").at(-2) || "."
      if (node.kind !== "user" && changing(node.id)) node.live = true
    })
    const legend = el("div", "legend")
    legend.append(
      append(el("span"), legendLine(false), document.createTextNode("Solid: the code confirms it")),
      append(el("span"), legendLine(true), document.createTextNode("Dashed: the code does not confirm it")),
    )
    const why = inline(
      el("p", "muted explain"),
      "To check an arrow, OpenCode reads the import lines of both files, such as `import { total } from \"./cart.js\"`, which shows that one file uses the other. " +
        "A dashed arrow means neither file imports the other: the flowchart says they work together, but the code shows no direct link. " +
        "It can still be right, for example when files talk through a web request or an event, so check that step. " +
        "Arrows from or to you, and arrows inside one file, have nothing to check.",
    )
    const area = el("div", "machine")
    area.dataset.nodes = JSON.stringify(nodes)
    area.dataset.steps = JSON.stringify(steps)
    const list = el("ol", "sequence")
    steps.forEach((step) => {
      const item = el("li")
      const name = (id) => nodes.find((node) => node.id === id).name
      item.append(document.createTextNode(name(step.from) + " → " + name(step.to) + ": "), inline(el("span"), step.action))
      if (step.confirmed === false) item.append(el("span", "warn-mark", "  (not confirmed: neither file imports the other)"))
      list.append(item)
    })
    const hint = el("p", "muted hint", "Scroll sideways to see the whole diagram.")
    hint.hidden = true
    const click = el("p", "muted explain", "Click a box, or the label of an arrow, to see what changed in that file next to the diagram.")
    const view = append(el("div", "flow" + (state.panel ? " open" : "")), append(el("div"), area, hint))
    if (state.panel) view.append(panel(entry, nodes, steps.find((step) => step.number === state.step)))
    section.append(legend, why, click, view, list)
    return section
  }

  // The side panel next to the flowchart: what changed in the file a box or an arrow points at, and why.
  function panel(entry, nodes, step) {
    const id = state.panel
    const box = el("aside", "panel")
    box.dataset.file = id
    box.setAttribute("aria-label", "What changed in " + id)
    const close = el("button", "close", "×")
    close.title = "Close"
    close.setAttribute("aria-label", "Close")
    close.addEventListener("click", hide)
    const name = append(el("div"), el("h3", "", id.split("/").at(-1)))
    if (id.includes("/")) name.append(el("div", "dir", "in " + id.slice(0, id.lastIndexOf("/") + 1)))
    box.append(append(el("header", "panel-head"), name, close))
    if (step) {
      const title = (node) => nodes.find((item) => item.id === node).name
      const text = append(el("span"), document.createTextNode(title(step.from) + " → " + title(step.to) + ": "), inline(el("span"), step.action))
      box.append(append(el("p", "step"), el("b", "", String(step.number)), text))
    }
    const record = entry.files.find((file) => file.file === id)
    const risk = (entry.affected || []).find((item) => item.file === id)
    const now = changing(id)
    if (!record && !risk && !now && !state.data.project.includes(id)) {
      box.append(el("p", "muted", "The flowchart names this file, but it is not in the project."))
      return box
    }
    const facts = el("div", "facts")
    if (now) facts.append(el("span", "badge live", "changing now"))
    if (record) facts.append(el("span", "badge " + record.status, STATUS[record.status]), counts(record.additions, record.deletions))
    if (!record && !now) facts.append(el("span", "badge " + (risk ? "affected" : "plain"), risk ? "may be affected" : "unchanged"))
    const tree = el("button", "toggle", "Show in the project tree")
    tree.addEventListener("click", () => pick(id, true))
    facts.append(tree)
    box.append(facts)
    // What the agent changed in the file since its last checkpoint, as it works.
    if (now) box.append(...diff(now, "live:" + id, undefined, true))
    if (!record && !risk && now) return box
    // A changed file can be explained as a whole, and each of its changes on its own.
    if (record) append(box, append(el("div", "tools"), explainer(entry, id)), state.shown.has(topic(entry, id)) ? explanation(entry, id) : undefined)
    const tools = (block, index) => ({
      buttons: [explainer(entry, id, index)],
      rows: state.shown.has(topic(entry, id, index)) ? [extra(explanation(entry, id, index))] : [],
    })
    // Links in the panel open the other file in the panel too.
    const go = (file) => (file === state.panel ? render() : show(file))
    box.append(...about(entry, id, go, "panel:" + key(entry) + ":" + id, tools))
    return box
  }

  // A file the agent changed since its last checkpoint, with its live diff, or undefined.
  function changing(file) {
    return state.data.unreviewed.find((item) => same(item.file, file))
  }

  // Which file or change of a checkpoint an explanation is about.
  function topic(entry, file, chunk) {
    return key(entry) + "|" + file + "|" + (chunk === undefined ? "file" : chunk)
  }

  function extra(content) {
    const cell = append(el("td"), content)
    cell.colSpan = 4
    return append(el("tr", "extra"), cell)
  }

  // The button that opens the model's explanation of a file, or of one of its changes, and closes it again.
  function explainer(entry, file, chunk) {
    const open = state.shown.has(topic(entry, file, chunk))
    const button = el("button", "tool" + (open ? " on" : ""), open ? "Hide the explanation" : chunk === undefined ? "Explain this file" : "Explain")
    button.addEventListener("click", () => {
      const target = topic(entry, file, chunk)
      if (open) state.shown.delete(target)
      else state.shown.add(target)
      if (!open && !state.threads.has(target)) return explain(entry, file, chunk)
      render()
    })
    return button
  }

  // Asks the map server for the model's explanation; nothing the model says is inserted as HTML.
  async function explain(entry, file, chunk, question) {
    const target = topic(entry, file, chunk)
    const item = { question, status: "asking", text: "" }
    state.threads.set(target, [...(state.threads.get(target) || []), item])
    render()
    const result = await post("explain", { number: entry.number, revision: entry.revision, file, chunk, question })
    item.status = result.ok ? "done" : "failed"
    item.text = result.ok ? result.body.answer : result.body.error || "The model did not answer. Try again."
    refresh()
  }

  // Sends a request to the map server. Requests that act are JSON from this page; the server refuses anything else.
  async function post(kind, body) {
    try {
      const response = await fetch(base + "/" + kind, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify(body),
      })
      return { ok: response.ok, body: await response.json() }
    } catch {
      return { ok: false, body: { error: "The map could not reach OpenCode. Is it still running?" } }
    }
  }

  // The model's explanations of a file or of one change, then a box to ask about what is still unclear.
  function explanation(entry, file, chunk) {
    const target = topic(entry, file, chunk)
    const box = el("div", "answer")
    ;(state.threads.get(target) || []).forEach((item) => {
      if (item.question) box.append(append(el("p"), el("b", "", "You asked: "), document.createTextNode(item.question)))
      if (item.status === "asking") box.append(el("p", "muted", "Asking the model…"))
      if (item.status === "failed") box.append(el("p", "failed", item.text))
      if (item.status === "done")
        box.append(append(el("div", "field said"), append(el("h4"), document.createTextNode("Explanation"), tag("claim", "the model's words · not verified")), inline(el("p"), item.text)))
    })
    const form = el("form", "ask")
    const input = el("input")
    input.type = "text"
    input.maxLength = 500
    input.placeholder = "What don't you understand?"
    input.setAttribute("aria-label", "What don't you understand?")
    input.value = state.drafts.get(target) || ""
    input.addEventListener("input", () => state.drafts.set(target, input.value))
    form.addEventListener("submit", (event) => {
      event.preventDefault()
      const question = input.value.trim()
      if (!question) return
      state.drafts.delete(target)
      explain(entry, file, chunk, question)
    })
    const send = el("button", "button", "Ask")
    send.type = "submit"
    box.append(append(form, input, send))
    return box
  }

  function legendLine(dashed) {
    const line = svg("svg", { viewBox: "0 0 30 10" })
    line.append(svg("line", { x1: "0", y1: "5", x2: "30", y2: "5", stroke: dashed ? "var(--wait)" : "currentColor", "stroke-width": "2", "stroke-dasharray": dashed ? "5 4" : "" }))
    return line
  }

  // Lays out the flowchart once the page has its width, as a sequence: each file is a column with its box on top, and
  // each step is a row of its own with a numbered arrow from one column to another. A label has its row to itself, so
  // it can never cover a box or another label; a longer label only makes its row taller.
  function layout() {
    document.querySelectorAll(".machine").forEach((area) => {
      const nodes = JSON.parse(area.dataset.nodes)
      const steps = JSON.parse(area.dataset.steps)
      // Columns shrink to fit the page; when even the narrowest are too wide, the diagram scrolls sideways.
      const column = Math.max(150, Math.min(240, (area.clientWidth - 32) / nodes.length))
      const width = Math.min(column - 24, 190)
      const height = nodes.some((node) => node.folder) ? 70 : 56
      // A diagram narrower than the page is centered in it.
      const offset = Math.max(16, (area.clientWidth - nodes.length * column) / 2)
      const center = (index) => offset + index * column + column / 2
      const at = new Map(nodes.map((node, index) => [node.id, index]))
      const stage = el("div", "stage")
      const drawing = svg("svg")
      const marker = svg("marker", { id: "tip", viewBox: "0 0 10 10", refX: "9", refY: "5", markerWidth: "7", markerHeight: "7", orient: "auto-start-reverse" })
      marker.append(svg("path", { d: "M0,0 L10,5 L0,10 z", fill: "currentColor" }))
      const defs = svg("defs")
      defs.append(marker)
      drawing.append(defs)
      // The picked arrow lights up both of its boxes; a picked box lights up on its own.
      const picked = steps.find((step) => step.number === state.step)
      const lit = (id) => (picked ? picked.from === id || picked.to === id : state.panel === id)
      const boxes = nodes.map((node, index) => {
        const box = el("button", "node " + node.kind + (lit(node.id) ? " lit" : "") + (node.live ? " live" : ""))
        box.dataset.id = node.id
        box.style.left = center(index) - width / 2 + "px"
        box.style.top = "16px"
        box.style.width = width + "px"
        box.style.height = height + "px"
        const parts = node.parts || []
        const word = { user: "starts it", missing: "not in the project", context: "unchanged", affected: "may be affected" }[node.kind] || STATUS[node.kind]
        box.title = (node.kind === "missing" ? node.id + " (not in the project)" : node.file || "You") + (parts.length ? "\nChanged: " + parts.join(", ") : "")
        append(box, el("span", "name", node.name))
        if (node.folder) box.append(el("span", "folder", "in " + node.folder + "/"))
        box.append(el("span", "kind", node.live ? "changing now" : word + (parts.length ? " · " + parts.join(", ") : "")))
        if (node.kind !== "user") box.addEventListener("click", () => show(node.id))
        return box
      })
      // A label over an arrow stays between the two columns it joins; the label of a step inside one file sits beside it.
      const labels = steps.map((step) => {
        const a = at.get(step.from)
        const b = at.get(step.to)
        const unchecked = step.confirmed === false
        const label = el("button", "label" + (unchecked ? " unchecked" : "") + (state.step === step.number ? " lit" : ""))
        if (unchecked) label.title = "Not confirmed by the code: neither file imports the other"
        append(label, el("b", "", String(step.number)), inline(el("span"), step.action))
        // An arrow opens the file it points at, or the file it starts from when it goes back to the user.
        const target = step.to === "user" ? step.from : step.to
        if (target !== "user") label.addEventListener("click", () => show(target, step.number))
        if (a === b) {
          label.style.left = center(a) + 40 + "px"
          label.style.maxWidth = Math.max(column - 52, 160) + "px"
        } else {
          label.style.left = (center(a) + center(b)) / 2 + "px"
          label.style.maxWidth = Math.min(Math.abs(center(b) - center(a)) - 16, 420) + "px"
          label.style.transform = "translateX(-50%)"
        }
        return label
      })
      append(stage, drawing, ...boxes, ...labels)
      area.replaceChildren(stage)
      // With their widths set, the labels' heights are known: each step gets a row as tall as its label, then its arrow.
      const first = 16 + height + 22
      const rows = steps.reduce((all, step, index) => {
        const y = all.length ? all[all.length - 1].next : first
        labels[index].style.top = y + "px"
        const arrow = y + labels[index].offsetHeight + 8
        const a = at.get(step.from)
        const b = at.get(step.to)
        return [...all, { a, b, arrow, next: arrow + (a === b ? 24 : 0) + 18 }]
      }, [])
      const bottom = rows.length ? rows[rows.length - 1].next : first
      nodes.forEach((_, index) =>
        drawing.append(svg("path", { d: `M${center(index)},${16 + height} V${bottom}`, class: "life" })),
      )
      rows.forEach((row, index) => {
        const unchecked = steps[index].confirmed === false
        const d =
          row.a === row.b
            ? `M${center(row.a)},${row.arrow} h28 v24 h-26`
            : `M${center(row.a)},${row.arrow} H${center(row.b) + (row.b > row.a ? -2 : 2)}`
        drawing.append(svg("path", { d, class: "edge" + (unchecked ? " unchecked" : ""), "marker-end": "url(#tip)" }))
      })
      const frame = stage.getBoundingClientRect()
      const right = Math.max(offset + nodes.length * column + 16, ...labels.map((label) => label.getBoundingClientRect().right - frame.left + 16))
      drawing.setAttribute("width", String(right))
      drawing.setAttribute("height", String(bottom))
      stage.style.width = right + "px"
      stage.style.height = bottom + "px"
      const hint = area.nextElementSibling
      if (hint && hint.classList.contains("hint")) hint.hidden = right <= area.clientWidth + 1
    })
  }

  // Files the checkpoint did not change that import a changed file: if the change broke what they rely on, they break too.
  function risks(entry) {
    const list = entry.affected || []
    if (!list.length) return undefined
    const section = part("May be affected", tag("fact", "found from the import lines"))
    section.append(
      el("p", "muted explain", "These files were not changed in this checkpoint, but they import a file that was. If the change broke something they rely on, they break too, so check them."),
      append(
        el("ul", "risks"),
        ...list.map((item) => {
          const link = el("button", "toggle", item.file)
          link.addEventListener("click", () => pick(item.file, true))
          return append(el("li"), link, document.createTextNode(item.names.length ? " uses " + sentence(mentioned(entry, item)) : " imports " + sentence(item.uses)))
        }),
      ),
    )
    return section
  }

  // The whole project as a tree, with the files this checkpoint changed marked. Clicking a file shows what changed and why.
  // Files that changed in other checkpoints look like any other file here, so each checkpoint shows only its own step.
  function project(entry, files) {
    const section = part("The project", tag("fact", "files and changes computed from the code"))
    section.id = "project"
    const changes = new Map(entry.files.map((file) => [file.file, file]))
    const risky = new Map((entry.affected || []).map((item) => [item.file, item]))
    // Files the agent is changing now are listed too, even new ones the project list does not have yet.
    const live = state.data.unreviewed.map((file) => file.file)
    const all = [...new Set([...files, ...changes.keys(), ...risky.keys(), ...live])].sort()
    const marked = all.filter((file) => changes.has(file) || risky.has(file) || live.includes(file))
    const shown = state.only ? marked : all
    const root = { folders: new Map(), files: [], path: "" }
    shown.forEach((file) => {
      let node = root
      file
        .split("/")
        .slice(0, -1)
        .forEach((part) => {
          const path = node.path + part + "/"
          if (!node.folders.has(part)) node.folders.set(part, { folders: new Map(), files: [], path, name: part })
          node = node.folders.get(part)
        })
      node.files.push(file)
    })
    const tree = el("div", "tree")
    const tools = el("label", "tree-tools")
    const box = el("input")
    box.type = "checkbox"
    box.checked = state.only
    box.addEventListener("change", () => {
      state.only = box.checked
      render()
    })
    append(tools, box, document.createTextNode("Show only changed files" + (risky.size ? " and files that may be affected" : "") + " (" + marked.length + ")"))
    tree.append(tools, branch(root, changes, risky, entry))
    if (files.length >= 3000) tree.append(el("p", "muted", "This project is large, so only part of it is listed."))
    const details = el("div", "details")
    detail(details, entry)
    section.append(append(el("div", "project"), tree, details))
    if (entry.unmatched && entry.unmatched.length) {
      section.append(
        append(
          el("div", "warn"),
          el("strong", "", "The agent wrote about files that have no changes in this checkpoint:"),
          append(el("ul"), ...entry.unmatched.map((note) => append(el("li"), el("code", "", note.file), document.createTextNode(" — " + note.change)))),
        ),
      )
    }
    return section
  }

  function branch(node, changes, risky, entry) {
    const holder = el("div")
    ;[...node.folders.values()]
      .sort((a, b) => a.name.localeCompare(b.name))
      .forEach((folder) => {
        const inside = [...changes.keys()].filter((file) => file.startsWith(folder.path))
        const near = [...risky.keys()].filter((file) => file.startsWith(folder.path) && !changes.has(file))
        const now = state.data.unreviewed.some((file) => file.file.startsWith(folder.path))
        const open = state.only || (state.folders.has(folder.path) ? state.folders.get(folder.path) : inside.length + near.length > 0 || now)
        const row = append(el("button", "row folder"), el("span", "caret", open ? "▾" : "▸"), el("span", "name", folder.name + "/"))
        const counted = [inside.length ? inside.length + " changed" : "", near.length ? near.length + " may be affected" : ""].filter(Boolean)
        if (counted.length) row.append(el("span", "count", counted.join(" · ")))
        row.setAttribute("aria-expanded", String(open))
        row.addEventListener("click", () => {
          state.folders.set(folder.path, !open)
          render()
        })
        holder.append(row)
        if (open) holder.append(append(el("div", "children"), branch(folder, changes, risky, entry)))
      })
    node.files.forEach((file) => {
      const change = changes.get(file)
      const row = el("button", "row" + (change || risky.has(file) || changing(file) ? "" : " plain") + (state.file === file ? " selected" : ""))
      append(row, el("span", "caret", ""), el("span", "name", file.split("/").at(-1)))
      if (changing(file)) row.append(el("span", "badge live", "changing now"))
      if (change) row.append(el("span", "badge " + change.status, STATUS[change.status]))
      if (risky.has(file)) row.append(el("span", "badge affected", "may be affected"))
      if (change) row.append(append(el("span", "count"), counts(change.additions, change.deletions)))
      row.title = file
      row.addEventListener("click", () => pick(file))
      holder.append(row)
      // The functions and classes this checkpoint changed in the file; each opens the diff where it changed.
      const symbols = (change && change.symbols) || []
      if (!symbols.length) return
      const parts = el("div", "children")
      symbols.slice(0, MAX_SYMBOLS).forEach((symbol, index) => {
        const item = append(el("button", "row symbol"), el("span", "caret", ""), el("span", "name", named(symbol)), el("span", "change " + symbol.change, CHANGES[symbol.change]))
        item.title = named(symbol) + (symbol.change === "removed" ? ", removed from lines " : ", lines ") + symbol.start + "–" + symbol.end
        item.addEventListener("click", () => focus(key(entry) + ":" + file, file, index, pick))
        parts.append(item)
      })
      if (symbols.length > MAX_SYMBOLS) parts.append(el("p", "muted more", symbols.length - MAX_SYMBOLS + " more in the file's details"))
      holder.append(parts)
    })
    return holder
  }

  // What changed in the picked file and why: the agent's note next to the lines from the diff.
  function detail(details, entry) {
    const file = state.file
    if (!file) return details.append(el("p", "muted", "Click a file to see what changed in it and why."))
    const record = entry.files.find((item) => item.file === file)
    const title = append(el("h3", "path"), document.createTextNode(file))
    if (record) title.append(el("span", "badge " + record.status, STATUS[record.status]), counts(record.additions, record.deletions))
    details.append(title, ...about(entry, file, pick, key(entry) + ":" + file))
  }

  // A file in one checkpoint: the agent's note, the functions it changed, why it may break, the files around it, and
  // its changed lines. The project view and the flowchart's panel both show it: `go` opens another file in the same
  // place, `id` keeps the state of its diff, and `tools` adds the panel's buttons to each change.
  function about(entry, file, go, id, tools) {
    const record = entry.files.find((item) => item.file === file)
    const risk = (entry.affected || []).find((item) => item.file === file)
    const items = []
    if (!record && !risk) items.push(el("p", "muted", "Not changed in this checkpoint."))
    if (record && record.note) items.push(note(record.note))
    if (record && !record.note) items.push(el("p", "muted", "The agent wrote no note for this file."))
    if (record && record.symbols && record.symbols.length) items.push(parts(record.symbols, id, file, go))
    if (risk) items.push(warning(risk, entry, go))
    const links = entry.links || []
    const near = (entry.affected || []).filter((item) => item.uses.includes(file)).map((item) => item.file)
    const uses = [...new Set(links.filter((link) => link.from === file).map((link) => link.to))]
    const users = [...new Set(links.filter((link) => link.to === file).map((link) => link.from))].filter((item) => !near.includes(item))
    if (uses.length || users.length || near.length) {
      const relations = el("div", "relations")
      const chips = (list) => list.map((item) => {
        const chip = el("button", "toggle", item)
        chip.addEventListener("click", () => go(item))
        return chip
      })
      if (uses.length) relations.append(append(el("span"), document.createTextNode("Uses"), ...chips(uses)))
      if (users.length) relations.append(append(el("span"), document.createTextNode("Used by"), ...chips(users)))
      if (near.length) relations.append(append(el("span"), document.createTextNode("Unchanged files that use it"), ...chips(near)))
      items.push(relations)
    }
    if (record) items.push(...diff(record, id, tools))
    return items
  }

  // The agent's note about a file: claims, shown next to the computed diff.
  function note(text) {
    return append(
      el("div", "field note"),
      append(el("h4"), document.createTextNode("The agent's note"), tag("claim")),
      inline(append(el("p"), el("b", "", "What it is: ")), text.purpose),
      inline(append(el("p"), el("b", "", "What changed and why: ")), text.change),
    )
  }

  // The functions and classes the change touched in a file; each opens the diff where it changed.
  function parts(symbols, id, file, go) {
    const chips = el("div", "chips")
    symbols.forEach((symbol, index) => {
      const chip = append(el("button", "chip"), document.createTextNode(named(symbol)), el("span", "change " + symbol.change, CHANGES[symbol.change]))
      chip.addEventListener("click", () => focus(id, file, index, go))
      chips.append(chip)
    })
    return append(el("div", "field"), append(el("h4"), document.createTextNode("Functions and classes it changed"), tag("fact", "from the code")), chips)
  }

  // Why a file the checkpoint did not change may still break, with the lines where it uses what changed.
  function warning(item, entry, go) {
    const text = el("p")
    text.append(document.createTextNode("Not changed in this checkpoint, but it imports "))
    item.uses.forEach((used, index) => {
      if (index) text.append(document.createTextNode(index === item.uses.length - 1 ? " and " : ", "))
      const link = el("button", "toggle", used)
      link.addEventListener("click", () => go(used))
      text.append(link)
    })
    text.append(document.createTextNode(", which changed."))
    if (item.names.length) text.append(document.createTextNode(" It uses " + sentence(mentioned(entry, item)) + ", so check that it still works with the new code."))
    const table = el("table")
    item.lines.forEach((line) => table.append(append(el("tr"), el("td", "n", String(line.line)), el("td", "t", line.text))))
    return append(
      el("div", "risk"),
      append(el("div", "field"), append(el("h4"), document.createTextNode("May be affected"), tag("fact", "from the import lines"))),
      text,
      item.lines.length ? append(el("div", "lines"), table) : undefined,
    )
  }

  // The changed lines change by change, each under the functions it falls in; the lines around them open on request.
  // The project view shows the first few changed lines. The flowchart's panel shows every change, with `tools` giving
  // its buttons and the rows they opened.
  function diff(file, id, tools, live) {
    const open = state.open.has(id)
    const changed = file.blocks.flatMap((block) => block.lines.filter((line) => line.kind !== "same"))
    const heading = live
      ? append(el("div", "field"), append(el("h4"), document.createTextNode("Changing now"), tag("fact", "from the agent's edits · not in a checkpoint yet")))
      : append(el("div", "field"), append(el("h4"), document.createTextNode(open ? "The full diff" : "What was added and removed"), tag("fact", "from the diff")))
    if (!file.blocks.length) return [heading, el("p", "muted", "No line-by-line diff was saved for this file.")]
    const limit = tools ? Infinity : KEY_LINES
    const table = el("table")
    let room = limit
    file.blocks.forEach((block, index) => {
      if (!open && room <= 0) return
      const lines = open ? block.lines : block.lines.filter((line) => line.kind !== "same").slice(0, room)
      room -= lines.length
      const added = tools ? tools(block, index) : { buttons: [], rows: [] }
      table.append(where(block, file, id, added.buttons))
      lines.forEach((line) => table.append(row(line)))
      added.rows.forEach((item) => table.append(item))
    })
    const result = [heading, append(el("div", "lines"), table)]
    if (open || changed.length > limit || file.blocks.some((block) => block.lines.some((line) => line.kind === "same"))) {
      const more = el("button", "toggle", open ? "Show only the changed lines" : "Show the full diff (" + plural(changed.length, "changed line") + ")")
      more.addEventListener("click", () => {
        if (open) state.open.delete(id)
        else state.open.add(id)
        render()
      })
      result.push(more)
    }
    return result
  }

  // The header of one change: the functions it falls in and the lines it changed, with the panel's buttons.
  function where(block, file, id, buttons) {
    const names = (block.symbols || []).map((index) => named(file.symbols[index]))
    const marked = state.mark && state.mark.id === id && (block.symbols || []).includes(state.mark.index)
    const text = names.length ? "In " + sentence(names) + " · " + block.span : block.span.charAt(0).toUpperCase() + block.span.slice(1)
    const cell = el("td")
    cell.colSpan = 4
    if (buttons.length) cell.append(append(el("div", "where-row"), el("span", "", text), append(el("span", "tools"), ...buttons)))
    else cell.textContent = text
    return append(el("tr", "where" + (marked ? " mark" : "")), cell)
  }

  function row(line) {
    return append(
      el("tr", line.kind),
      el("td", "n", line.old === undefined ? "" : String(line.old)),
      el("td", "n", line.new === undefined ? "" : String(line.new)),
      el("td", "s", line.kind === "add" ? "+" : line.kind === "del" ? "−" : " "),
      el("td", "t", line.text),
    )
  }

  function did(entry) {
    const section = part("What the agent did", tag("claim"))
    section.append(append(el("ol", "steps"), ...entry.steps.map((step) => inline(el("li"), step.replace(/^\s*(?:\d+[.)]|[-*•])\s+/, "")))))
    const decision = el("div", "decision")
    decision.append(el("strong", "", "Your decision: "), document.createTextNode(DECISIONS[status(entry)]))
    if (entry.comment) decision.append(document.createTextNode(" — “" + entry.comment + "”"))
    section.append(decision)
    return section
  }

  function session(stats, list) {
    const section = part("This session")
    const figures = [
      [list.length, "checkpoints"],
      [stats.checkpoints.approved + stats.checkpoints.auto, "approved"],
      [stats.checkpoints.revised, "revisions asked"],
      [stats.checkpoints.stopped, "stopped"],
      [stats.checkpoints.forced, "forced by the safety net"],
      [stats.checkpoints.sentBack, "explanations sent back to the agent"],
      [duration(stats.run.waitingMs), "your review time"],
      [duration(stats.run.totalMs), "total time"],
      [stats.changes.files, "files changed"],
    ]
    section.append(append(el("div", "figures"), ...figures.map((item) => append(el("div", "figure"), el("strong", "", String(item[0])), el("span", "", item[1])))))
    const download = el("a", "button", "Download study data (JSON)")
    download.href = base + "/stats"
    download.setAttribute("download", "")
    section.append(download)
    return section
  }

  function live(on) {
    document.getElementById("dot").className = "dot " + (on ? "on" : "off")
    document.getElementById("status").textContent = on
      ? "Live · updated " + new Date().toLocaleTimeString()
      : "Not connected. Is OpenCode still running?"
  }

  async function poll() {
    try {
      const response = await fetch(base + "/data", { cache: "no-store" })
      if (!response.ok) throw new Error(String(response.status))
      const raw = await response.text()
      live(true)
      if (raw !== state.raw) {
        state.raw = raw
        state.data = JSON.parse(raw)
        refresh()
      }
    } catch {
      live(false)
    }
    setTimeout(poll, 1500)
  }

  document.addEventListener("keydown", (event) => {
    if (event.key !== "Escape") return
    if (state.chat && document.activeElement && document.activeElement.closest("#chat")) return toggleChat(false)
    if (state.panel && !typing()) hide()
  })
  document.getElementById("chat-open").addEventListener("click", () => toggleChat(!state.chat))
  document.getElementById("chat-close").addEventListener("click", () => toggleChat(false))
  document.getElementById("chat-form").addEventListener("submit", (event) => {
    event.preventDefault()
    ask()
  })
  // Enter sends; Shift+Enter starts a new line.
  document.getElementById("chat-text").addEventListener("keydown", (event) => {
    if (event.key !== "Enter" || event.shiftKey || event.isComposing) return
    event.preventDefault()
    ask()
  })
  // An update that waited for the user to stop typing shows once they leave the box. The short wait lets a click on a
  // button next to the box land before the page is redrawn.
  document.addEventListener("focusout", () =>
    setTimeout(() => {
      if (state.stale && !typing()) render()
    }, 300),
  )
  // The diagram's columns depend on its width, so it is laid out again when the window changes size.
  let sizing = false
  window.addEventListener("resize", () => {
    if (sizing) return
    sizing = true
    requestAnimationFrame(() => {
      sizing = false
      layout()
    })
  })
  poll()
})()
